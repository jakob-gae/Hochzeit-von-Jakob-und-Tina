const SPREADSHEET_ID = "1aFeHuceMVupURE9hLvzwWWGZqKt6GUF2-awFw70o9Hk"; const SHEET_NAME = "Hochzeits-Rückmeldungen"; const TIME_ZONE = "Europe/Berlin"; const MAX_PERSONS = 20; // Spalten (Reihenfolge = Reihenfolge in der Tabelle) const HEADERS = [ "Eingegangen am", // A "Antwort", // B "Name", // C "Essenswünsche & Unverträglichkeiten", // D "Nachricht", // E "Antwortgruppe" // F ]; const VERSION = "2026-11-16-v4-geschenke"; // ========================================== // VERSIONS-CHECK: Web-App-URL im Browser öffnen // ========================================== function doGet(e) { try { if (e && e.parameter && e.parameter.action === "gifts") { return json_({ ok: true, gifts: listGifts_() }); } return json_({ ok: true, version: VERSION, hinweis: "Rückmelde-Skript ist aktiv." }); } catch (err) { console.error(err); return json_({ ok: false, error: String(err.message || err) }); } } // ========================================== // HAUPTFUNKTION: WEBSITE-RÜCKMELDUNGEN // ========================================== function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { throw new Error("Keine Formulardaten empfangen."); } const data = JSON.parse(e.postData.contents); // Spam-Schutz (Honeypot) if (data.website) { return json_({ ok: true, spam: true }); } // Geschenkeliste: Reservieren / Freigeben if (data.action === "reserveGift") return reserveGift_(data); if (data.action === "releaseGift") return releaseGift_(data); // Antwort prüfen const answer = String(data.attendance || "").trim().toLowerCase(); if (answer !== "yes" && answer !== "no") { throw new Error("Ungültige Rückmeldung: " + JSON.stringify(data.attendance)); } const isYes = answer === "yes"; const attendance = isYes ? "Zusage" : "Absage"; // Personen prüfen const attendees = Array.isArray(data.attendees) ? data.attendees : []; if (attendees.length === 0) { throw new Error("Bitte mindestens eine Person mit Namen angeben."); } if (attendees.length > MAX_PERSONS) { throw new Error("Zu viele Personen in einer Rückmeldung."); } const timestamp = new Date(); const groupId = Utilities.getUuid(); // Nachricht nur bei Absage const message = isYes ? "" : clean_(data.message); const rows = attendees.map((person, index) => { const name = clean_(person && person.name); if (!name) { throw new Error("Bei jeder Person muss ein Name angegeben werden."); } // Essenswünsche nur bei Zusage (alte Felder food/allergies werden weiter unterstützt) const food = isYes ? clean_( person.foodAndAllergies || [person.food, person.allergies].filter(Boolean).join("; ") ) : ""; return [ timestamp, // A attendance, // B name, // C food, // D index === 0 ? message : "", // E (nur einmal pro Rückmeldung) groupId // F ]; }); // Alles Schreiben unter Lock, auch das Anlegen der Überschriften const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const sheet = getSheet_(); const firstRow = nextFreeRow_(sheet); // Spalten B–F als Text formatieren: verhindert Formel-Auswertung // und automatische Umwandlungen (Datum, Zahl) sheet.getRange(firstRow, 2, rows.length, HEADERS.length - 1).setNumberFormat("@"); sheet.getRange(firstRow, 1, rows.length, 1).setNumberFormat("dd.MM.yyyy HH:mm:ss"); sheet.getRange(firstRow, 1, rows.length, HEADERS.length).setValues(rows); SpreadsheetApp.flush(); } finally { lock.releaseLock(); } return json_({ ok: true, count: rows.length, answer: attendance, groupId: groupId }); } catch (err) { console.error(err); return json_({ ok: false, error: String(err.message || err) }); } } // ========================================== // GESCHENKELISTE (anonym) // ========================================== // Blatt "Geschenkeliste" mit den Spalten: // A ID | B Wunsch | C Beschreibung | D Preis | E Link | F Bild | G Status | H Freigabe-Hash // Es wird nie gespeichert, WER reserviert hat. Spalte H enthält nur einen // Hash des Freigabecodes, den der Gast beim Reservieren erhält. const GIFT_SHEET = "Geschenkeliste"; const GIFT_HEADERS = ["ID", "Wunsch", "Beschreibung", "Preis", "Link", "Bild", "Status", "Freigabe-Hash"]; const GIFT_TAKEN = "reserviert"; const GIFT_FREE = "frei"; // EINMALIG im Editor ausführen: legt das Blatt mit Beispielzeilen an function setupGifts() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); if (ss.getSheetByName(GIFT_SHEET)) { Logger.log("Das Blatt '" + GIFT_SHEET + "' existiert bereits."); return; } const sheet = ss.insertSheet(GIFT_SHEET); sheet.getRange(1, 1, 1, GIFT_HEADERS.length).setValues([GIFT_HEADERS]).setFontWeight("bold"); sheet.setFrozenRows(1); sheet.getRange(2, 8, sheet.getMaxRows() - 1, 1).setNumberFormat("@"); sheet.getRange(2, 7, sheet.getMaxRows() - 1, 1).setDataValidation( SpreadsheetApp.newDataValidation().requireValueInList([GIFT_FREE, GIFT_TAKEN], true).build() ); sheet.getRange(2, 1, 2, GIFT_HEADERS.length).setValues([ ["", "Beispiel: Kaffeemaschine", "Für gemütliche Sonntagsfrühstücke", "ca. 80 €", "https://www.example.com", "bilder/kaffeemaschine.jpg", GIFT_FREE, ""], ["", "Beispiel: Bettwäsche", "Weiß oder Naturtöne, 135×200", "ca. 50 €", "", "", GIFT_FREE, ""] ]); sheet.setColumnWidth(2, 200); sheet.setColumnWidth(3, 280); sheet.setColumnWidth(5, 220); sheet.setColumnWidth(6, 200); sheet.hideColumns(8); } function getGiftSheet_() { const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getSheetByName(GIFT_SHEET); if (!sheet) throw new Error("Das Blatt '" + GIFT_SHEET + "' fehlt. Bitte setupGifts() ausführen."); return sheet; } function hashCode_(code) { const bytes = Utilities.computeDigest( Utilities.DigestAlgorithm.SHA_256, String(code).trim().toUpperCase() ); return bytes.map(b => ("0" + (b & 0xff).toString(16)).slice(-2)).join(""); } function newGiftCode_() { const alphabet = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // ohne 0/O, 1/I let code = ""; for (let i = 0; i < 6; i++) { code += alphabet.charAt(Math.floor(Math.random() * alphabet.length)); } return code; } // Liste für die Webseite (ohne Hashes). Fehlende IDs werden automatisch ergänzt. function listGifts_() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const sheet = ss.getSheetByName(GIFT_SHEET); if (!sheet || sheet.getLastRow() < 2) return []; const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const n = sheet.getLastRow() - 1; const values = sheet.getRange(2, 1, n, GIFT_HEADERS.length).getValues(); let changed = false; values.forEach(r => { if (String(r[1]).trim() && !String(r[0]).trim()) { r[0] = Utilities.getUuid().slice(0, 8); changed = true; } }); if (changed) { sheet.getRange(2, 1, n, 1).setValues(values.map(r => [r[0]])); } return values .filter(r => String(r[1]).trim()) .map(r => ({ id: String(r[0]).trim(), name: String(r[1]).trim(), description: String(r[2]).trim(), price: String(r[3]).trim(), link: String(r[4]).trim(), image: String(r[5]).trim(), taken: String(r[6]).trim().toLowerCase() === GIFT_TAKEN })); } finally { lock.releaseLock(); } } function reserveGift_(data) { const id = String(data.id || "").trim(); if (!id) throw new Error("Kein Wunsch angegeben."); const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const sheet = getGiftSheet_(); if (sheet.getLastRow() < 2) throw new Error("Wunsch nicht gefunden."); const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, GIFT_HEADERS.length).getValues(); const idx = values.findIndex(r => String(r[0]).trim() === id); if (idx < 0) throw new Error("Wunsch nicht gefunden."); if (String(values[idx][6]).trim().toLowerCase() === GIFT_TAKEN) { throw new Error("Dieser Wunsch wurde gerade schon von jemand anderem reserviert."); } // eindeutigen Code erzeugen const existing = values.map(r => String(r[7])); let code, hash; do { code = newGiftCode_(); hash = hashCode_(code); } while (existing.indexOf(hash) >= 0); const row = idx + 2; sheet.getRange(row, 8).setNumberFormat("@"); sheet.getRange(row, 7, 1, 2).setValues([[GIFT_TAKEN, hash]]); SpreadsheetApp.flush(); return json_({ ok: true, code: code }); } finally { lock.releaseLock(); } } function releaseGift_(data) { const code = String(data.code || "").trim(); if (!code) throw new Error("Bitte einen Freigabecode angeben."); const hash = hashCode_(code); const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const sheet = getGiftSheet_(); if (sheet.getLastRow() < 2) throw new Error("Dieser Code ist nicht bekannt."); const values = sheet.getRange(2, 1, sheet.getLastRow() - 1, GIFT_HEADERS.length).getValues(); const idx = values.findIndex(r => String(r[7]) === hash && String(r[6]).trim().toLowerCase() === GIFT_TAKEN ); if (idx < 0) throw new Error("Dieser Code ist nicht bekannt oder bereits freigegeben."); sheet.getRange(idx + 2, 7, 1, 2).setValues([[GIFT_FREE, ""]]); SpreadsheetApp.flush(); return json_({ ok: true }); } finally { lock.releaseLock(); } } // ========================================== // TABELLENBLATT HOLEN / ANLEGEN // ========================================== // Stimmen die Überschriften in Zeile 1 nicht mit HEADERS überein // (z. B. alte Struktur), wird das alte Blatt umbenannt (nichts geht verloren) // und ein frisches Blatt mit korrekten Spalten angelegt. function getSheet_() { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (sheet && sheet.getLastRow() > 0) { const first = sheet.getRange(1, 1, 1, HEADERS.length).getValues()[0]; const matches = HEADERS.every((h, i) => String(first[i]).trim() === h); if (!matches) { const stamp = Utilities.formatDate(new Date(), TIME_ZONE, "yyyy-MM-dd HH-mm"); sheet.setName(SHEET_NAME + " (alt " + stamp + ")"); sheet = null; } } if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]).setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; } // ========================================== // NÄCHSTE FREIE ZEILE (anhand Spalte A) // ========================================== // getLastRow() zählt jede beschriftete Zelle im Blatt mit (z. B. Notizen in // Spalte H) und würde neue Zeilen zu weit unten einfügen. function nextFreeRow_(sheet) { const values = sheet.getRange(1, 1, sheet.getMaxRows(), 1).getValues(); for (let i = values.length - 1; i >= 0; i--) { if (values[i][0] !== "" && values[i][0] !== null) { return i + 2; } } return 2; } // ========================================== // EINGABEN BEREINIGEN // ========================================== function clean_(value) { return String(value == null ? "" : value).trim().slice(0, 1000); // Formelschutz erfolgt über das Textformat "@" der Zielzellen } // ========================================== // JSON-ANTWORT // ========================================== function json_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ========================================== // TESTS (im Editor auswählen und "Ausführen") // ========================================== function testZusage() { const result = doPost({ postData: { contents: JSON.stringify({ attendance: "yes", attendees: [ { name: "Testperson Zusage 1", foodAndAllergies: "Vegetarisch, keine Nüsse" }, { name: "Testperson Zusage 2", foodAndAllergies: "" } ], message: "", website: "" }) } }); Logger.log(result.getContent()); } function testAbsage() { const result = doPost({ postData: { contents: JSON.stringify({ attendance: "no", attendees: [ { name: "Testperson Absage 1", foodAndAllergies: "" }, { name: "Testperson Absage 2", foodAndAllergies: "" } ], message: "Leider können wir nicht kommen.", website: "" }) } }); Logger.log(result.getContent()); }
